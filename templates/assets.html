How to use in JSON

    "images": { "cell_diagram": "biology/cell_structure.png" },
    "svgs":   { "circuit":      "physics/circuit.svg" },
    "videos": { "pendulum":     "physics/pendulum.mp4" }
  },
  "scenes": [{
    "type": "concept",
    "steps": [
      {
        "audio": "Look at this diagram...",
        "render": { "action": "show", "target": "image", "src": "cell_diagram" }
      },
      {
        "audio": "Watch this clip...",
        "render": { "action": "show", "target": "video_clip", "src": "pendulum",
                    "caption": "Simple Pendulum Motion" }
      }
    ]
  }]
}
